cmake_minimum_required(VERSION 3.24)
project(lsdware VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
add_library(lsd_core src/core.cpp src/config.cpp src/source2.cpp)
target_include_directories(lsd_core PUBLIC include)
target_link_libraries(lsd_core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(lsd_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(lsd_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(lsd_tests tests/tests.cpp)
  target_link_libraries(lsd_tests PRIVATE lsd_core)
  foreach(suite math pattern geometry config schema profile entities)
    add_test(NAME ${suite} COMMAND lsd_tests ${suite} ${CMAKE_CURRENT_SOURCE_DIR}/data)
  endforeach()
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Only Windows x64 is supported")
  endif()
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG f5befd2d29e66809cd1110a152e375a7f1981f06)
  FetchContent_MakeAvailable(imgui)
  add_library(imgui_lib STATIC ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(imgui_lib PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_compile_definitions(imgui_lib PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_link_libraries(imgui_lib PUBLIC d3d11 dxgi d3dcompiler dwmapi)
  add_executable(lsdware src/main.cpp src/process_win.cpp src/overlay_win.cpp)
  target_compile_definitions(lsdware PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(lsdware PRIVATE lsd_core imgui_lib dcomp user32 gdi32 ole32)
  if(MSVC)
    target_compile_options(lsdware PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(lsdware PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(lsdware PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  add_custom_command(TARGET lsdware POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/data $<TARGET_FILE_DIR:lsdware>/data)
endif()
